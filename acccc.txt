' --------------------------------------------------
' 配列内の要素を、指定した列数でグリッド状に配置する
'
' 使用例:
'   $grid(["a", "b", "c", "d", "e"], 3)
'
' 配置:
'   a  b  c
'   d  e
' --------------------------------------------------
!unquoted procedure $grid($items, $columns)

    !assert %get_json_type($items) == "array" : "$items must be an array"
    !assert $columns > 0 : "$columns must be greater than 0"

    !$count = %size($items)
    !$index = 0

    !while $index < $count

        !$current = $items[$index]
        !$column = %mod($index, $columns)

        ' 右隣の要素
        !$rightIndex = $index + 1

        !if ($column < $columns - 1) && ($rightIndex < $count)
            !$right = $items[$rightIndex]
            $current -right[hidden]-> $right
        !endif

        ' 同じ列の次の行
        !$belowIndex = $index + $columns

        !if $belowIndex < $count
            !$below = $items[$belowIndex]
            $current -down[hidden]-> $below
        !endif

        !$index = $index + 1

    !endwhile

!endprocedure
