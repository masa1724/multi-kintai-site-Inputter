' =====================================================
' 共通設定
'
' 複数の引数・レコードは「~」で区切る。
' プロシージャ定義・呼び出しは1物理行で記述する。
' =====================================================


' -----------------------------------------------------
' 表示用の空行
'
' 末尾の単純な改行は省略される場合があるため、
' 全角スペースを含む行を空行として使用する。
' -----------------------------------------------------

!$blankLine = "\n<size:10>　</size>"


' =====================================================
' ジョブ生成プロシージャ
'
' $id          : PlantUML上の識別子
' $feature     : 機能名
' $jobName     : ジョブ名
' $programName : プログラム名
' $args        : 実行引数。「~」区切り。省略可能
' $cycle       : 日次、月次、期次、年次、随時
' $kind        : normal、sameProgram
'
' $argsが空の場合、[args]は表示しない。
' プログラム名の下には常に空行を表示する。
' =====================================================

!procedure $job($id, $feature, $jobName, $programName, $args="", $cycle="日次", $kind="normal")

    ' ---------------------------------------------
    ' 機能バッジ
    ' ---------------------------------------------

    !$featureBg = "#F1F5F9"
    !$featureFg = "#475569"

    !if $feature == "取込"
        !$featureBg = "#E0F2FE"
        !$featureFg = "#0369A1"
    !elseif $feature == "統合・分類"
        !$featureBg = "#F3E8FF"
        !$featureFg = "#7E22CE"
    !elseif $feature == "集計"
        !$featureBg = "#DCFCE7"
        !$featureFg = "#15803D"
    !elseif $feature == "反映"
        !$featureBg = "#FEF3C7"
        !$featureFg = "#B45309"
    !elseif $feature == "検証"
        !$featureBg = "#F1F5F9"
        !$featureFg = "#475569"
    !endif


    ' ---------------------------------------------
    ' 実行周期
    ' ---------------------------------------------

    !$cycleColor = "#2563EB"
    !$cycleMark = "●"

    !if $cycle == "月次"
        !$cycleColor = "#059669"
        !$cycleMark = "■"
    !elseif $cycle == "期次"
        !$cycleColor = "#D97706"
        !$cycleMark = "◆"
    !elseif $cycle == "年次"
        !$cycleColor = "#7C3AED"
        !$cycleMark = "★"
    !elseif $cycle == "随時"
        !$cycleColor = "#64748B"
        !$cycleMark = "○"
    !endif


    ' ---------------------------------------------
    ' ヘッダ
    ' ---------------------------------------------

    !$featureBadge = "<back:" + $featureBg + "><color:" + $featureFg + "><b> " + $feature + " </b></color></back>"
    !$cycleBadge = "<color:" + $cycleColor + "><b>" + $cycleMark + " " + $cycle + "</b></color>"

    !$label = "<size:9>" + $featureBadge + "  " + $cycleBadge

    !if $kind == "sameProgram"
        !$label = $label + "  <back:#EEF2FF><color:#4F46E5><b> 共通PG </b></color></back>"
    !endif

    !$label = $label + "</size>"


    ' ---------------------------------------------
    ' ジョブ名
    ' ---------------------------------------------

    !$label = $label + "\n<b>" + $jobName + "</b>"


    ' ---------------------------------------------
    ' プログラム物理名
    '
    ' argsの有無にかかわらず、
    ' 物理名の下に空行を表示する。
    ' ---------------------------------------------

    !if $programName != ""
        !$label = $label + "\n<size:10><color:#475569>" + $programName + "</color></size>"
        !$label = $label + $blankLine
    !endif


    ' ---------------------------------------------
    ' 実行引数
    ' ---------------------------------------------

    !if $args != ""
        !$argList = %splitstr($args, "~")

        !$label = $label + "\n<size:10><color:#475569><b>[args]</b></color></size>\l"

        !foreach $arg in $argList
            !$label = $label + "<size:10><color:#64748B>- " + $arg + "</color></size>\l"
        !endfor
    !endif

    rectangle "$label" as $id
!endprocedure


' =====================================================
' テーブル生成プロシージャ
'
' $id           : PlantUML上の識別子
' $logicalName  : 論理テーブル名
' $physicalName : 物理テーブル名
' $records      : レコード一覧。「~」区切り。省略可能
' $kind         : normal、result
'
' $recordsが空の場合、[records]は表示しない。
' 物理名の下には常に空行を表示する。
' =====================================================

!procedure $table($id, $logicalName, $physicalName, $records="", $kind="normal")

    ' ---------------------------------------------
    ' 論理名
    ' ---------------------------------------------

    !$label = "<b>" + $logicalName + "</b>"


    ' ---------------------------------------------
    ' 物理名
    '
    ' recordsの有無にかかわらず、
    ' 物理名の下に空行を表示する。
    ' ---------------------------------------------

    !if $physicalName != ""
        !$label = $label + "\n<size:10><color:#475569>" + $physicalName + "</color></size>"
        !$label = $label + $blankLine
    !endif


    ' ---------------------------------------------
    ' レコード一覧
    ' ---------------------------------------------

    !if $records != ""
        !$recordList = %splitstr($records, "~")

        !$label = $label + "\n<size:10><color:#475569><b>[records]</b></color></size>\l"

        !foreach $record in $recordList
            !$label = $label + "<size:10><color:#64748B>- " + $record + "</color></size>\l"
        !endfor
    !endif


    ' ---------------------------------------------
    ' テーブル生成
    ' ---------------------------------------------

    !if $kind == "result"
        database "$label" as $id <<result>>
    !else
        database "$label" as $id
    !endif
!endprocedure


' =====================================================
' ファイル生成プロシージャ
'
' $id           : PlantUML上の識別子
' $logicalName  : 論理ファイル名
' $physicalName : 物理ファイル名
' $records      : レコード一覧。「~」区切り。省略可能
'
' $recordsが空の場合、[records]は表示しない。
' 物理名の下には常に空行を表示する。
' =====================================================

!procedure $dataFile($id, $logicalName, $physicalName, $records="")

    ' ---------------------------------------------
    ' 論理名
    ' ---------------------------------------------

    !$label = "<b>" + $logicalName + "</b>"


    ' ---------------------------------------------
    ' 物理名
    '
    ' recordsの有無にかかわらず、
    ' 物理名の下に空行を表示する。
    ' ---------------------------------------------

    !if $physicalName != ""
        !$label = $label + "\n<size:10><color:#475569>" + $physicalName + "</color></size>"
        !$label = $label + $blankLine
    !endif


    ' ---------------------------------------------
    ' レコード一覧
    ' ---------------------------------------------

    !if $records != ""
        !$recordList = %splitstr($records, "~")

        !$label = $label + "\n<size:10><color:#475569><b>[records]</b></color></size>\l"

        !foreach $record in $recordList
            !$label = $label + "<size:10><color:#64748B>- " + $record + "</color></size>\l"
        !endfor
    !endif

    file "$label" as $id
!endprocedure
